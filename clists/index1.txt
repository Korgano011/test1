<script>
// test-api listens on 127.0.0.1:3000 (CORS on), so this works from file:// or any local page
const SERVER = 'http://localhost:3000';
const API = SERVER + '/christmas_lists';
const SMS_NUMBER = '7022045387';
let currentRows = [];

async function loadItems() {
  try {
    const res = await fetch(API);
    if (!res.ok) throw new Error((await res.json()).error);
    currentRows = await res.json();
  } catch (err) {
    currentRows = [];
    document.getElementById('status').textContent = 'Server is down - new items will be texted to ' + SMS_NUMBER;
  }
  renderTable();
}

// CQL text that gets texted when the server is down (single quotes doubled for CQL)
function insertCql(name, item) {
  const q = v => "'" + String(v).replace(/'/g, "''") + "'";
  return 'insert into christmas_lists (name, item)\nvalues (' + q(name) + ', ' + q(item) + ');';
}

function textInsert(name, item) {
  const body = insertCql(name, item);
  const link = 'sms:' + SMS_NUMBER + '&body=' + encodeURIComponent(body);
  document.getElementById('smsText').value = body;
  document.getElementById('smsLink').href = link;
  document.getElementById('smsBox').style.display = 'block';
  document.getElementById('status').textContent = 'Server is down - texting the insert to ' + SMS_NUMBER;
  window.location.href = link;
}

document.getElementById('copySms').addEventListener('click', () => {
  const box = document.getElementById('smsText');
  box.select();
  navigator.clipboard.writeText(box.value).catch(() => document.execCommand('copy'));
});

function renderTable() {
  const search = document.getElementById('search').value.toLowerCase();
  const list = document.getElementById('items');
  list.innerHTML = '';
  currentRows.filter(row =>
    (row.name || '').toLowerCase().includes(search) ||
    row.item.toLowerCase().includes(search)
  ).forEach(row => {
    const tr = document.createElement('tr');
    [row.name, row.item].forEach(value => {
      const td = document.createElement('td');
      td.textContent = value;
      tr.appendChild(td);
    });
    const buttonCell = document.createElement('td');
    const editButton = document.createElement('button');
    editButton.textContent = 'Edit';
    editButton.addEventListener('click', () => editRow(tr, row));
    buttonCell.appendChild(editButton);
    const deleteButton = document.createElement('button');
    deleteButton.textContent = 'Delete';
    deleteButton.addEventListener('click', () => deleteItem(row.item));
    buttonCell.appendChild(deleteButton);
    tr.appendChild(buttonCell);
    list.appendChild(tr);
  });
}

function editRow(tr, row) {
  tr.innerHTML = '';
  const nameInput = document.createElement('input');
  nameInput.value = row.name || '';
  const itemInput = document.createElement('input');
  itemInput.value = row.item;
  [nameInput, itemInput].forEach(input => {
    const td = document.createElement('td');
    td.appendChild(input);
    tr.appendChild(td);
  });
  const buttonCell = document.createElement('td');
  const saveButton = document.createElement('button');
  saveButton.textContent = 'Save';
  saveButton.addEventListener('click', () => updateItem(row.item, nameInput.value, itemInput.value));
  buttonCell.appendChild(saveButton);
  const cancelButton = document.createElement('button');
  cancelButton.textContent = 'Cancel';
  cancelButton.addEventListener('click', loadItems);
  buttonCell.appendChild(cancelButton);
  tr.appendChild(buttonCell);
}

async function updateItem(oldItem, name, item) {
  const res = await fetch(API + '/' + encodeURIComponent(oldItem), {
    method: 'PUT',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ name, item })
  });
  const data = await res.json();
  document.getElementById('status').textContent = res.ok ? 'Updated ' + data.item : 'Error: ' + data.error;
  loadItems();
}

async function deleteItem(item) {
  const res = await fetch(API + '/' + encodeURIComponent(item), { method: 'DELETE' });
  const data = await res.json();
  document.getElementById('status').textContent = res.ok ? 'Deleted ' + item : 'Error: ' + data.error;
  loadItems();
}

document.getElementById('search').addEventListener('input', renderTable);

document.getElementById('deleteAll').addEventListener('click', async () => {
  if (!confirm('Delete every row in the table?')) return;
  const res = await fetch(API, { method: 'DELETE' });
  const data = await res.json();
  document.getElementById('status').textContent = res.ok ? 'Deleted all rows' : 'Error: ' + data.error;
  loadItems();
});

document.getElementById('savePdf').addEventListener('click', () => {
  const doc = new jspdf.jsPDF();
  doc.text('Christmas Wish List Items', 14, 15);
  doc.autoTable({
    startY: 20,
    head: [['Name', 'Item']],
    body: currentRows.map(row => [row.name, row.item])
  });
  doc.save('ChristmasLists.pdf');
});

document.getElementById('itemForm').addEventListener('submit', async (e) => {
  e.preventDefault();
  const name = document.getElementById('name').value.trim();
  const item = document.getElementById('item').value.trim();
  document.getElementById('smsBox').style.display = 'none';
  let res, data;
  try {
    res = await fetch(API, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ name, item })
    });
    data = await res.json();
  } catch (err) {
    // API container unreachable
    textInsert(name, item);
    e.target.reset();
    return;
  }
  if (res.status >= 500) {
    // API is up but Cassandra is not
    textInsert(name, item);
    e.target.reset();
    return;
  }
  document.getElementById('status').textContent = res.ok ? 'Saved ' + data.item : 'Error: ' + data.error;
  e.target.reset();
  loadItems();
});

let lastQuery = null;

document.getElementById('saveQueryPdf').addEventListener('click', () => {
  if (!lastQuery) return;
  const doc = new jspdf.jsPDF();
  doc.setFontSize(14);
  doc.text('Problem solving result', 14, 15);
  doc.setFontSize(10);
  const queryLines = doc.splitTextToSize('Query: ' + lastQuery.cql, 180);
  doc.text(queryLines, 14, 23);
  const y = 23 + queryLines.length * 5;
  doc.text('Run: ' + lastQuery.ranAt.toLocaleString(), 14, y);
  doc.autoTable({
    startY: y + 5,
    head: [lastQuery.columns],
    body: lastQuery.rows.map(r => r.map(v => v === null ? '' : String(v)))
  });
  doc.save('query-result.pdf');
});

document.getElementById('queryForm').addEventListener('submit', async (e) => {
  e.preventDefault();
  const status = document.getElementById('queryStatus');
  const table = document.getElementById('queryResult');
  const pdfButton = document.getElementById('saveQueryPdf');
  table.innerHTML = '';
  lastQuery = null;
  pdfButton.disabled = true;
  status.textContent = 'Running...';
  try {
    const res = await fetch(SERVER + '/query', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ cql: document.getElementById('cql').value })
    });
    const data = await res.json();
    if (!res.ok) { status.textContent = 'Error: ' + data.error; return; }
    status.textContent = data.rows.length + ' row(s)';
    lastQuery = { cql: document.getElementById('cql').value.trim(), columns: data.columns, rows: data.rows, ranAt: new Date() };
    pdfButton.disabled = false;
    const head = document.createElement('tr');
    data.columns.forEach(c => {
      const th = document.createElement('th');
      th.textContent = c;
      head.appendChild(th);
    });
    table.appendChild(head);
    data.rows.forEach(r => {
      const tr = document.createElement('tr');
      r.forEach(v => {
        const td = document.createElement('td');
        td.textContent = v === null ? '' : v;
        tr.appendChild(td);
      });
      table.appendChild(tr);
    });
  } catch (err) {
    status.textContent = 'Error: ' + err.message;
  }
});

loadItems();
</script>
